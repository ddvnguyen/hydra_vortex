```go
// upload.go — package uploader (abridged)
package uploader

import (
    "context"
    "os"

    "github.com/aws/aws-sdk-go/aws"
    "github.com/aws/aws-sdk-go/service/s3/s3manager"
)

type Uploader struct {
    Up  *s3manager.Uploader
    Log func(msg string, args ...any)
}

// Upload reads the whole file into RAM, then uploads. OOM-killed at 1.2 GB RSS.
func (u *Uploader) Upload(ctx context.Context, bucket, key, path string) error {
    body, err := os.ReadFile(path) // 5 GB file -> OOM on the small node
    if err != nil {
        return err
    }
    _, err = u.Up.UploadWithContext(ctx, &s3manager.UploadInput{
        Bucket: aws.String(bucket),
        Key:    aws.String(key),
        Body:   bytes.NewReader(body),
    })
    return err
}
```

`s3manager` notes (vendored README excerpt):
```
UploadInput.Body may be an io.Reader; the manager buffers PART SIZE per concurrent
part when the body is not an io.ReadSeeker with a known size. Use an *os.File (it
is a ReadSeeker) so parts stream from disk. Concurrency N with part size S holds
~N*S in flight plus SDK overhead (~8 MB per part). Min part size 5 MB (S3 API
floor); all parts except the last must be >= 5 MB or S3 returns EntityTooSmall.
A completed multipart upload needs CompleteMultipartUpload; an aborted run leaves
parts billable until AbortMultipartUpload or the bucket lifecycle rule fires.
```
