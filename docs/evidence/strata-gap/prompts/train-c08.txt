`wasm/src/lib.rs` exposes `#[wasm_bindgen] pub fn score(bytes: &[u8]) -> f32` but large inputs copy twice (JS→wasm linear memory→Vec) and 10 MB payloads take 900 ms. Rework to accept `Uint8Array` view with `js_sys`, process in 64 KB chunks without materializing, return a struct `{ score, n }` via `JsValue::from_serde`, and add a `wasm-pack test --node` case asserting chunked == whole-input score on a 1 MB fixture. Note the one lifetime pitfall with the borrowed view.

use std::sync::{Arc, Mutex};
use std::time::{Duration, Instant};
use tokio::sync::mpsc;

const CAP: usize = 3418;
const MAX_ATTEMPTS: u32 = 5;

pub struct Scorer {
    tx: mpsc::Sender<Vec<u8>>,
    inner: Arc<Mutex<BatchState>>,
}

struct BatchState {
    buf: Vec<Vec<u8>>,
    dropped: u64,
    last_flush: Option<Instant>,
}

impl Scorer {
    pub fn new() -> (Self, mpsc::Receiver<Vec<u8>>) {
        let (tx, rx) = mpsc::channel(CAP);
        let s = Scorer { tx, inner: Arc::new(Mutex::new(BatchState {
            buf: Vec::new(), dropped: 0, last_flush: None })) }};
        (s, rx)
    }

    pub async fn score(&self, msg: Vec<u8>) -> Result<(), SendError> {
        let mut attempt = 0u32;
        loop {
            match self.tx.try_send(msg.clone()) {
                Ok(()) => return Ok(()),
                Err(mpsc::error::TrySendError::Full(m)) => {
                    attempt += 1;
                    if attempt >= MAX_ATTEMPTS { return Err(SendError::Overload); }
                    let backoff = Duration::from_millis(50 * (1 << attempt));
                    tokio::time::sleep(backoff).await;
                    let _ = m;
                }
                Err(mpsc::error::TrySendError::Closed(_)) => return Err(SendError::Closed),
            }
        }
    }
}
