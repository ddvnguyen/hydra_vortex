const cache = new Map<string, Promise<string>>();

export function Farm(file: File) {
  const [progress, setProgress] = useState(0);
  useEffect(() => {
    FarmUpload(file).then((url) => {
      setProgress(100);  // BUG: no cleanup — fires after unmount
      return url;
    }});
  }, [file]);
  return progress;
}

async function FarmUpload(f: File): Promise<string> {
  const h = await hash(f);
  if (!cache.has(h)) cache.set(h, post(f));
  return cache.get(h)!;
}
