```rust
// decode.rs — packet field decoder (abridged)
use std::slice;

pub struct Ctx<'a> {
    pub buf: &'a [u8],
    pub off: usize,
}

#[inline]
pub fn field_len(hdr: &[u8]) -> usize {
    // network order u16 at hdr[2..4]; NO range check against the packet
    u16::from_be_bytes([hdr[2], hdr[3]]) as usize
}

// line 112 — the crash site:
pub unsafe fn read_field(ptr: *const u8, len: usize) -> &'static [u8] {
    // SAFETY: caller guarantees len bytes are mapped (it does not)
    slice::from_raw_parts(ptr, len)
}

pub fn decode(ctx: &mut Ctx) -> Result<Frame, Error> {
    let hdr = &ctx.buf[ctx.off..ctx.off + 8];
    let len = field_len(hdr);
    let ptr = unsafe { ctx.buf.as_ptr().add(ctx.off + 8) };
    let body = unsafe { read_field(ptr, len) }; // panics when len > avail
    ctx.off += 8 + len;
    Frame::parse(body)
}
```
