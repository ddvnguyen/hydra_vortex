Bug report (verbatim):
```
# Summary field not being set when using path() method

## Description

When using the `path()` method in APISpec to define a path with a summary,
the summary field is not being set in the resulting OpenAPI specification.
This issue affects both OpenAPI 2.0 and 3.0.0 specifications.

The `description` field works correctly, but the `summary` field is completely
ignored.

## How to Reproduce

from apispec import APISpec

spec = APISpec(title="Pets API", version="1.0.0", openapi_version="3.0.0")
spec.path(path="/pet/{petId}", summary="Operations on a Pet",
           description="Operations on a Pet identified by its ID")
# resulting spec has 'description' but no 'summary' under /pet/{petId}
```

import pytest

@pytest.mark.parametrize("version", ["2.0", "3.0.0"])
def test_path_summary(spec_factory, version):
    spec = spec_factory(openapi_version=version)
    spec.path("/pet/{petId}", summary="Operations on a Pet",
              description="Operations on a Pet identified by its ID")
    op = spec.to_dict()["paths"]["/pet/{petId}"]["get"]
    assert op["summary"] == "Operations on a Pet"
    assert op["description"] == "Operations on a Pet identified by its ID"

@pytest.mark.parametrize("version", ["2.0", "3.0.0"])
def test_path_summary_only_no_empty_description(spec_factory, version):
    spec = spec_factory(openapi_version=version)
    spec.path("/pet/{petId}", summary="Operations on a Pet")
    op = spec.to_dict()["paths"]["/pet/{petId}"]["get"]
    assert op["summary"] == "Operations on a Pet"
    assert "description" not in op

def test_path_summary_with_explicit_ops_wins(spec_factory):
    # DOCUMENTED CHOICE: explicit operations= wins over the summary= kwarg.
    spec = spec_factory(openapi_version="3.0.0")
    spec.path("/pet/{petId}", summary="kwarg summary",
              operations={"get": {"summary": "explicit summary"}})
    op = spec.to_dict()["paths"]["/pet/{petId}"]["get"]
    assert op["summary"] == "explicit summary"
