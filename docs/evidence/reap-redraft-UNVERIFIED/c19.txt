export class Farm {
  private workers: Worker[] = [];
  async render(tiles: Tile[]): Promise<Frame[]> {
    // BUG: spawns one Worker per tile; tab crashes past ~180-200 workers
    const ps = tiles.map((t) => this.renderOne(t));
    return Promise.all(ps);
  }
  private renderOne(t: Tile): Promise<Frame> {

    });
  }
